  const title = track?.title || (loading ? 'Ищем ваш звук…' : 'Музыка скоро появится');

    <section className="featured-release" aria-label="В фокусе">

            type="button"

import { useState, type ReactNode } from 'react';
import type { Track } from '../types/track';

type Props = {
  track?: Track | null;
  className?: string;
  variant?: 'mobile' | 'desktop';
  children?: ReactNode;
  eager?: boolean;
};

export function TrackArtwork({ track, className, variant = 'desktop', children, eager }: Props) {
  const [failedUrl, setFailedUrl] = useState<string>();
  const url = track?.coverUrl;
  const coverClass = variant === 'desktop'
    ? 'du-cover'
    : 'relative flex flex-shrink-0 items-center justify-center overflow-hidden bg-surface';

  return (
    <span className={className ? `${coverClass} ${className}` : coverClass}
      data-tone={(track?.id ?? 0) % 4} aria-hidden="true">
      {variant === 'desktop' && <span className="du-mini-disc" />}
      {url && url !== failedUrl ? (
        <img
          key={url}
          src={url}
          loading={eager ? 'eager' : 'lazy'}
          onError={() => setFailedUrl(url)}
          className={variant === 'mobile' ? 'w-full h-full object-cover' : undefined}
          alt=""
        />
      ) : variant === 'mobile' && (
        <span className="text-xl text-muted opacity-50">♪</span>
      )}
      {children}
    </span>
  );
}

import { Icon, IconButton, time } from './Controls';
import { TrackArtwork } from './TrackArtwork';
import type { PlayerStatus, Track } from '../types/track';

type Props = {
  variant: 'mobile' | 'desktop';
  track: Track;
  isActive: boolean;
  status: PlayerStatus;
  isFavorite: boolean;
  favoritePending?: boolean;
  index?: number;
  onPlay: (track: Track) => void;
  onToggleFavorite: (track: Track) => void;
};

export function TrackRow({
  variant, track, isActive, status, isFavorite, favoritePending, index, onPlay, onToggleFavorite,
}: Props) {
  const isPlaying = isActive && status === 'playing';
  const isLoading = isActive && status === 'loading';
  const title = `${track.title} — ${track.artist}`;
  const playLabel = `${isPlaying ? 'Приостановить' : 'Слушать'} ${title}`;
  const favoriteLabel = `${isFavorite ? 'Убрать из избранного' : 'В избранное'}: ${track.title}`;
  const duration = track.duration ? time(track.duration) : undefined;

  if (variant === 'desktop') {
    return (
      <div className={`du-trackrow${isActive ? ' active' : ''}`} role="listitem">
        <button type="button" className="du-trackpick" aria-label={playLabel}
          onClick={() => onPlay(track)}>
          <span className="du-num" aria-hidden="true">
            {isPlaying ? <b>♫</b> : index === undefined ? null : String(index + 1).padStart(2, '0')}
          </span>
          <TrackArtwork track={track} />
          <span className="min-w-0">
            <strong className="block truncate">{track.title}</strong>
            <small className="block truncate">{track.artist}</small>
          </span>
        </button>
        <span className="du-album truncate">{track.album || '—'}</span>
        <span className="du-duration">{duration ?? '—'}</span>
        <IconButton icon="heart" size={18} className={isFavorite ? 'du-liked' : ''}
          aria-pressed={isFavorite} disabled={favoritePending} label={favoriteLabel}
          onClick={() => onToggleFavorite(track)} />
      </div>
    );

  return (
    <div className={`flex items-center gap-1.5 px-4 py-2.5 shadow-xl/20 transition-colors animate-fade-up ${isActive ? 'bg-surface/40' : ''}`}>
      <button type="button" aria-label={playLabel} aria-busy={isLoading || undefined}
        className="flex min-w-0 flex-1 items-center gap-3 rounded-xl text-left active:bg-surface/60"
        onClick={() => onPlay(track)}>
        <TrackArtwork track={track} variant="mobile" className="h-12 w-12 rounded-xl hover:opacity-80">
          {isActive && (
            <span className="absolute inset-0 flex items-center justify-center bg-black/40 text-white">
              {isLoading
                ? <span className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin-slow" />
                : <Icon name={isPlaying ? 'pause' : 'play'} size={16} />}
            </span>
          )}
        </TrackArtwork>
        <span className="flex-1 min-w-0">
          <span className={`block truncate text-[15px] font-medium transition-colors ${isActive ? 'text-accent' : 'text-text'}`}>
            {track.title}
          </span>
          <span className="block truncate text-[13px] text-muted font-mono">
            {track.artist}{track.album ? ` — ${track.album}` : ''}
          </span>
        </span>
        {isPlaying ? (
          <span className="flex items-end gap-[2px] h-4 mr-1" aria-hidden="true">

      }}

    </dialog>
  );

import { useCallback, useEffect, useRef, useState } from 'react';
import { getFavorites, toggleFavorite as saveFavorite } from '../../services/api';
import type { Track } from '../../types/track';

type FavoritesState = {
  token: string | null;
  tracks: Track[];
  pending: Set<number>;
  status: 'loading' | 'ready' | 'error';
  notice: string;
};

function initial(token: string | null): FavoritesState {
  return { token, tracks: [], pending: new Set(), status: token ? 'loading' : 'ready', notice: '' };
}

export function useFavorites(
  token: string | null,
  openAuth: () => void,
  checkError: (error: unknown, token: string) => void,
) {
  const [state, setState] = useState(() => initial(token));
  const [attempt, setAttempt] = useState(0);
  const session = useRef<AbortController>();
  const pending = useRef(new Set<number>());
  const clearNotice = () => setState((old) => ({ ...old, notice: '' }));
  const reloadFavorites = () => setAttempt((value) => value + 1);

  useEffect(() => {
    const controller = new AbortController();
    session.current = controller;
    pending.current.clear();
    setState(initial(token));
    if (token) {
      getFavorites(token, controller.signal)
        .then((tracks) => {
          if (!controller.signal.aborted) setState({ ...initial(token), tracks, status: 'ready' });
        })
        .catch((error) => {
          if (controller.signal.aborted) return;
          checkError(error, token);
          setState({ ...initial(token), status: 'error' });
        });
    }
    return () => controller.abort();
  }, [token, attempt, checkError]);

  const visible = state.token === token ? state : initial(token);
  const toggleFavorite = useCallback(async (track: Track) => {
    if (!token) {
      openAuth();
      return;
    }
    const { id } = track;
    const controller = session.current;
    if (!controller || state.token !== token || state.status !== 'ready' || pending.current.has(id)) return;
    const previous = state.tracks.find((favorite) => favorite.id === id);
    pending.current.add(id);
    setState((old) => ({
      ...old, pending: new Set(pending.current),
      tracks: previous ? old.tracks.filter((favorite) => favorite.id !== id) : [track, ...old.tracks],
    }));
    try {
      await saveFavorite(token, id, Boolean(previous));
    } catch (error) {
      if (controller.signal.aborted) return;
      checkError(error, token);
      setState((old) => ({
        ...old,
        tracks: previous ? [previous, ...old.tracks] : old.tracks.filter((favorite) => favorite.id !== id),
        notice: 'Не удалось изменить избранное. Попробуйте ещё раз.',
      }));
    } finally {
      if (!controller.signal.aborted) {
        pending.current.delete(id);
        setState((old) => ({ ...old, pending: new Set(pending.current) }));
      }
    }
  }, [token, state, openAuth, checkError]);

  return {
    favorites: visible.tracks,
    favoriteIds: new Set(visible.tracks.map((track) => track.id)),
    pendingIds: visible.pending,
    canToggle: !token || visible.status === 'ready',
    favoritesLoading: visible.status === 'loading',
    favoritesError: visible.status === 'error' ? 'Не удалось загрузить избранное.' : '',
    reloadFavorites, toggleFavorite, notice: visible.notice, clearNotice,
  };
}

import type { Track } from '../../types/track';

export type RepeatMode = 'none' | 'all' | 'one';

export interface PlaybackQueue {
  tracks: Track[];
  index: number;
  order: number[] | null;
  shuffle: boolean;
  repeat: RepeatMode;
}

export const emptyQueue: PlaybackQueue = {
  tracks: [],
  index: -1,
  order: null,
  shuffle: false,
  repeat: 'none',
};

function shuffledOrder(length: number, first: number) {
  const remaining = Array.from({ length }, (_, index) => index).filter(index => index !== first);
  for (let end = remaining.length - 1; end > 0; end--) {
    const swap = Math.floor(Math.random() * (end + 1));
    [remaining[end], remaining[swap]] = [remaining[swap], remaining[end]];
  }
  return [first, ...remaining];
}

export function selectTrack(queue: PlaybackQueue, track: Track, trackList = queue.tracks): PlaybackQueue {
  const index = trackList.findIndex(item => item.id === track.id);
  const tracks = index === -1 ? [...trackList, track] : trackList;
  const selected = index === -1 ? tracks.length - 1 : index;
  return {
    ...queue,
    tracks,
    index: selected,
    order: queue.shuffle ? shuffledOrder(tracks.length, selected) : null,
  };
}

export function nextTrack(queue: PlaybackQueue, fromEnded: boolean): PlaybackQueue | null {
  if (!queue.tracks.length) return null;
  const position = queue.order ? queue.order.indexOf(queue.index) : queue.index;
  if (position + 1 < queue.tracks.length) {
    return { ...queue, index: queue.order?.[position + 1] ?? position + 1 };
  }
  if (fromEnded && queue.repeat !== 'all') return null;
  const index = queue.order?.[0] ?? 0;
  return {
    ...queue,
    index,
    order: queue.shuffle ? shuffledOrder(queue.tracks.length, index) : null,
  };
}

export function previousTrack(queue: PlaybackQueue): PlaybackQueue | null {
  if (!queue.tracks.length) return null;
  const position = queue.order ? queue.order.indexOf(queue.index) : queue.index;
  const previous = Math.max(0, position - 1);
  return { ...queue, index: queue.order?.[previous] ?? previous };
}

export function toggleShuffle(queue: PlaybackQueue): PlaybackQueue {
  const shuffle = !queue.shuffle;
  return {
    ...queue,
    shuffle,
    order: shuffle && queue.index >= 0 ? shuffledOrder(queue.tracks.length, queue.index) : null,
  };
}

const nextRepeat: Record<RepeatMode, RepeatMode> = { none: 'all', all: 'one', one: 'none' };

export function cycleRepeat(queue: PlaybackQueue): PlaybackQueue {
  return { ...queue, repeat: nextRepeat[queue.repeat] };
}

export function TelegramAccount({ profile, variant }: { profile: Profile; variant: Variant }) {
  const { initData, token } = useAccount();
  const labels = copy[variant];
  if (initData && !profile.hasPassword) return <MiniAppLinkForm variant={variant} initData={initData} />;
  if (!initData && token && profile.hasPassword) {

function WebsiteLinkForm({ variant, token, linked }: { variant: Variant; token: string; linked: boolean }) {
  const { changeAccount, checkError, refreshProfile } = useAccount();
  const [code, setCode] = useState<{ code: string; expiresAt: string } | null>(null);
  const [password, setPassword] = useState('');
  const [busy, setBusy] = useState(false);
  const [message, setMessage] = useState('');
  const [error, setError] = useState('');
  const labels = copy[variant];
  const desktop = variant === 'desktop';

  useEffect(() => {
    if (!code || linked) return;
    const controller = new AbortController();
    const expiresAt = Date.parse(code.expiresAt);
    let timer: number;

    async function checkLink() {
      try {
        if ((await refreshProfile(controller.signal))?.telegramId) return;
      } catch {
        // Retry transient profile errors while the code can still be used.
      }
      if (controller.signal.aborted) return;
      if (Date.now() >= expiresAt) {
        setCode(null);
        setMessage('');
        setError(labels.expired);
      } else {
        timer = window.setTimeout(checkLink, 2500);
      }
    }

    timer = window.setTimeout(checkLink, 2500);
    return () => {
      controller.abort();
      window.clearTimeout(timer);
    };
  }, [code, linked, refreshProfile, labels.expired]);

  useEffect(() => {
    if (!linked) return;
    setCode(null);
    setMessage('');
    setError('');
  }, [linked]);

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    if (!password || busy) return;
    setBusy(true);
    setError('');
    setMessage('');
    let disconnected = false;
    try {
      if (linked) {
        await changeAccount(async () => {
          const replacement = await unlinkTelegram(token, password);
          disconnected = true;
          return replacement;
        });
      } else {
        setCode(await createTelegramLinkCode(token, password));
      }
      setPassword('');
    } catch (cause) {
      checkError(cause, token);
      const wrongPassword = cause instanceof ApiError && cause.status === 400;
      setError(disconnected ? labels.disconnected : wrongPassword ? labels.wrongPassword : labels.failed);
    } finally {
      setBusy(false);
    }
  }

  async function copyCode() {
    if (!code) return;
    try {
      await navigator.clipboard.writeText(code.code);
      setMessage(labels.copied);
    } catch {
      setMessage(labels.select);
    }
  }

  return (
    <section className={'shrink-0 rounded-2xl border p-4 text-sm ' + (desktop ? desktopBox : mobileBox)}>
      {linked && <p role="status" className="mb-2 font-medium">{labels.connected}</p>}
      <details className="grid gap-2">
        <summary className="cursor-pointer font-semibold">{linked ? labels.disconnect : labels.connect}</summary>
        {code && !linked ? (
          <div className="mt-3 grid gap-2">
            <p className="opacity-80">{labels.instruction}</p>
            <code className="select-all break-all font-mono text-sm">{code.code}</code>
            <button type="button" className={desktop ? 'du-primary' : mobileButton} onClick={copyCode}>
              {labels.copy}
            </button>
          </div>
        ) : (
          <form className="mt-3 grid gap-2" onSubmit={submit}>
            {linked && <p className="opacity-80">{labels.unlinkHelp}</p>}
            <label className="grid gap-1">
              {labels.password}
              <input
                className={desktop ? 'du-input' : mobileInput}
                type="password"
                autoComplete="current-password"
                value={password}
                onChange={(event) => setPassword(event.target.value)}
                required
              />
            </label>
            <button className={desktop ? 'du-primary' : mobileButton} disabled={busy || !password}>
              {busy ? labels.waiting : linked ? labels.disconnect : labels.create}
            </button>
          </form>
        )}
        {message && <p className="mt-2" role="status">{message}</p>}
        {error && <p className="mt-2 text-danger" role="alert">{error}</p>}
      </details>
    </section>
  );
}

function MiniAppLinkForm({ variant, initData }: { variant: Variant; initData: string }) {
  const { changeAccount } = useAccount();
  const [code, setCode] = useState('');
  const [preview, setPreview] = useState<TelegramLinkPreview | null>(null);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');
  const labels = copy[variant];
  const desktop = variant === 'desktop';
  const enteredCode = code.trim();
  const validCode = /^[A-Za-z0-9_-]{24}$/.test(enteredCode);

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    if (busy || !validCode) return;
    setBusy(true);
    setError('');
    let confirmed = false;
    try {
      if (!preview) {
        setPreview(await previewTelegramLink(enteredCode, initData));
        return;
      }
      await changeAccount(async () => {
        const token = await confirmTelegramLink(enteredCode, initData, preview.mergeRequired);
        confirmed = true;
        return token;
      });
    } catch (cause) {
      if (confirmed) {
        setPreview(null);
        setCode('');
        setError(labels.saved);
      } else if (cause instanceof ApiError && cause.status === 401) setError(labels.reopen);
      else if (cause instanceof ApiError && cause.status === 404) setError(labels.expired);
      else if (cause instanceof ApiError && cause.status === 409) setError(preview ? labels.changed : labels.protected);
      else setError(labels.failed);
      if (cause instanceof ApiError && (cause.status === 404 || cause.status === 409)) setPreview(null);
    } finally {
      setBusy(false);
    }
  }

  return (
    <details className={'shrink-0 rounded-2xl border p-4 text-sm ' + (desktop ? desktopBox : mobileBox)}>
      <summary className="cursor-pointer font-semibold">{labels.enter}</summary>
      <form className="mt-3 grid gap-2" onSubmit={submit}>
        {preview ? (
          <>
            <p>{labels.join(preview.username)}</p>
            {preview.mergeRequired && <p className="opacity-80">{labels.mergeHelp}</p>}
          </>
        ) : (
          <>
            <p className="opacity-80">{labels.enterCode}</p>
            <label className="grid gap-1">
              {labels.code}
              <input
                className={desktop ? 'du-input' : mobileInput}
                autoComplete="one-time-code"
                autoCapitalize="none"
                spellCheck={false}
                value={code}
                onChange={(event) => setCode(event.target.value)}
                required
              />
            </label>
          </>
        )}
        {error && <p className="text-danger" role="alert">{error}</p>}
        <button className={desktop ? 'du-primary' : mobileButton} disabled={busy || !validCode}>
          {busy ? labels.waiting : preview ? preview.mergeRequired ? labels.merge : labels.confirm : labels.review}
        </button>
        {preview && (
          <button type="button" disabled={busy} onClick={() => { setPreview(null); setError(''); }}>
            {labels.back}
          </button>
        )}
      </form>
    </details>
  );
}

import { useEffect, useMemo, useState } from 'react';
import { searchTracks } from '../../services/api';
import type { Track } from '../../types/track';

type Result = { query: string; revision: number; tracks: Track[]; error: string };

export function useCatalogSearch(catalog: Track[], catalogRevision: number) {
  const [query, setQuery] = useState('');
  const [result, setResult] = useState<Result>();
  const [revision, setRevision] = useState(0);
  const term = query.trim();

  useEffect(() => {
    if (!term) return;
    const controller = new AbortController();
    const timeout = window.setTimeout(async () => {
      try {
        const tracks = await searchTracks(term, controller.signal);
        if (!controller.signal.aborted) setResult({ query: term, revision, tracks, error: '' });
      } catch {
        if (!controller.signal.aborted)
          setResult({ query: term, revision, tracks: [], error: 'Не удалось выполнить поиск.' });
      }
    }, 280);
    return () => {
      window.clearTimeout(timeout);
      controller.abort();
    };
  }, [term, revision, catalogRevision]);

  const current = result?.query === term && result.revision === revision ? result : undefined;
  const tracks = useMemo(() => {
    if (!term) return catalog;
    const matches = new Map(current?.tracks.map((track) => [track.id, track]));
    // Older servers search titles and artists only. The full catalog also supplies album matches.
    const needle = term.toLocaleLowerCase();
    for (const track of catalog) {
      if (track.album?.toLocaleLowerCase().includes(needle)) matches.set(track.id, track);
    }
    return Array.from(matches.values());
  }, [catalog, current, term]);

  return {
    query, setQuery, tracks, loading: Boolean(term && !current), error: current?.error ?? '',
    retry: () => setRevision((value) => value + 1),
  };
}

function countTracks(count: number) {
  const forms: Record<string, string> = { one: 'трек', few: 'трека', many: 'треков' };
  return count + ' ' + (forms[plural.select(count)] ?? 'треков');

function recentReleases(tracks: Track[]) {
  const artists = new Set<string>();
  const releases: Track[] = [];
  for (const track of tracks) {
    const artist = track.artist.trim().toLocaleLowerCase();
    if (artists.has(artist)) continue;
    releases.push(track);
    artists.add(artist);
    if (releases.length === 4) break;
  }
  return releases;
}

  session: {

    activeDeviceId: string | null;

export type PlaybackCommand =

  | { type: 'pause' | 'release'; position?: number }
  | { type: 'seek'; position: number }
  | { type: 'next' | 'previous' }
  | { type: 'ended'; trackId: number }
  | { type: 'setShuffle'; shuffle: boolean }
  | { type: 'setRepeat'; repeat: RepeatMode }
  | { type: 'transfer'; targetDeviceId: string };

export class PlaybackApiError extends ApiError {
  constructor(status: number, message: string, readonly snapshot?: PlaybackSnapshot) {
    super(status, message);

async function fetchPlayback(path: string, token: string, init: RequestInit) {
  const headers = new Headers(init.headers);
  headers.set('Authorization', `Bearer ${token}`);
  const result = await fetch(`${BASE_URL}/playback${path}`, { cache: 'no-store', ...init, headers });
  if (!result.ok) {
    const body = await result.json().catch(() => null);
    throw new PlaybackApiError(result.status, body?.error ?? `API error ${result.status}`, body?.snapshot);
  }
  return result;
}

export function playbackApi(token: string, credentials: DeviceCredentials) {
  async function post(path: string, payload: object, signal?: AbortSignal, keepalive?: boolean) {
    const result = await fetchPlayback(path, token, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ...credentials, ...payload }),
      signal,
      keepalive,
    });
    return result.json() as Promise<PlaybackSnapshot>;
  }

  return {
    register: (name: string, platform: DevicePlatform, signal: AbortSignal) => post('/devices', { name, platform }, signal),
    heartbeat: (signal: AbortSignal, report?: PlaybackReport) => post('/heartbeat', { report }, signal),
    command: (expectedVersion: number, command: PlaybackCommand, signal?: AbortSignal, keepalive?: boolean) =>
      post('/command', { expectedVersion, command }, signal, keepalive),
    acknowledge: (expectedVersion: number, position: number | undefined, signal: AbortSignal) =>
      post('/ack', { expectedVersion, position }, signal),
    session: async (signal: AbortSignal) => {
      const result = await fetchPlayback('/session', token, { signal });
      return result.json() as Promise<PlaybackSnapshot>;
    },
    events: (signal: AbortSignal, onSnapshot: (snapshot: PlaybackSnapshot) => void, onActivity: () => void) =>
      readPlaybackEvents(token, signal, onSnapshot, onActivity),
  };
}

async function readPlaybackEvents(
  token: string,
  signal: AbortSignal,
  onSnapshot: (snapshot: PlaybackSnapshot) => void,
  onActivity: () => void,
) {
  const result = await fetchPlayback('/events', token, { signal, headers: { Accept: 'text/event-stream' } });
  if (!result.body || !result.headers.get('Content-Type')?.includes('text/event-stream')) {
    throw new Error('Playback events unavailable');
  }
  const reader = result.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';

  try {
    while (!signal.aborted) {
      const { value, done } = await reader.read();
      if (done) break;
      onActivity();
      buffer += decoder.decode(value, { stream: true });
      const events = buffer.split(/\r?\n\r?\n/);
      buffer = events.pop() ?? '';
      for (const event of events) {
        const fields = event.split(/\r?\n/);
        if (!fields.some((field) => field.startsWith('event:') && field.slice(6).trim() === 'session')) continue;
        const data = fields.filter((field) => field.startsWith('data:')).map((field) => field.slice(5).trimStart());
        if (data.length) onSnapshot(JSON.parse(data.join('\n')) as PlaybackSnapshot);
      }
    }
  } finally {
    await reader.cancel().catch(() => {});
    reader.releaseLock();
  }
}

  return error instanceof ApiError && (error.status === 401 || error.status === 403);
}

export function useAuth(initData?: string) {
  const [session, setSession] = useState<Session>(checking);
  const [authOpen, setAuthOpen] = useState(false);
  const [notice, setNotice] = useState('');
  const [attempt, setAttempt] = useState(0);
  const current = useRef<AbortController>();
  const activeToken = useRef<string | null>(null);
  const accountChange = useRef<{ token: string | null } | null>(null);
  const openAuth = useCallback(() => setAuthOpen(true), []);
  const closeAuth = useCallback(() => setAuthOpen(false), []);
  const clearNotice = useCallback(() => setNotice(''), []);
  const retry = useCallback(() => setAttempt((value) => value + 1), []);

  const verify = useCallback(async (token: string, signal: AbortSignal, storedAtStart?: string | null) => {
    const profile = await getProfile(token, signal);
    if (signal.aborted) return;
    activeToken.current = token;
    if (storedAtStart === undefined || getToken() === storedAtStart) setToken(token);
    setSession({ status: 'authenticated', token, profile });
    setNotice('');
    setAuthOpen(false);
  }, []);

  useEffect(() => {
    accountChange.current = null;
    activeToken.current = null;
    const controller = new AbortController();
    current.current = controller;
    const { signal } = controller;
    const previous = getToken();
    setSession(checking);

    async function restore() {
      if (initData) {
        try {
          const token = await authWithTelegram(initData, signal);
          await verify(token, signal, previous);
          return;
        } catch (error) {
          if (signal.aborted) return;
          if (!previous && !invalidSession(error)) {
            setSession({ ...guest, status: 'unavailable' });
            return;
          }
        }
      }
      if (previous) {
        try {
          await verify(previous, signal, previous);
          return;
        } catch (error) {
          if (signal.aborted) return;
          if (!invalidSession(error) && !(error instanceof ApiError && error.status === 404)) {
            setSession({ ...guest, status: 'unavailable' });
            return;
          }
          if (getToken() === previous) clearToken();
          setNotice('Сессия истекла. Войдите снова.');
        }
      }
      setSession(guest);
    }

    void restore();
    return () => {
      accountChange.current = null;
      current.current?.abort();
    };
  }, [initData, attempt, verify]);

  const verifyCredentials = useCallback(async (token: string, formSignal?: AbortSignal) => {
    current.current?.abort();
    const controller = new AbortController();
    current.current = controller;
    const cancel = () => controller.abort();
    if (formSignal?.aborted) cancel();
    formSignal?.addEventListener('abort', cancel, { once: true });
    try {
      await verify(token, controller.signal);
    } finally {
      formSignal?.removeEventListener('abort', cancel);
    }
  }, [verify]);

  const acceptCredentials = useCallback((token: string, formSignal?: AbortSignal) => {
    accountChange.current = null;
    return verifyCredentials(token, formSignal);
  }, [verifyCredentials]);

  const changeAccount = useCallback(async (requestToken: () => Promise<string>) => {
    if (accountChange.current) throw new Error('Another account change is in progress');
    const change = { token: activeToken.current };
    accountChange.current = change;
    try {
      const token = await requestToken();
      if (accountChange.current === change && activeToken.current === change.token) {
        await verifyCredentials(token);
      }
    } finally {
      if (accountChange.current === change) accountChange.current = null;
    }
  }, [verifyCredentials]);

  const logout = useCallback(() => {
    accountChange.current = null;
    current.current?.abort();
    const token = activeToken.current;
    activeToken.current = null;
    if (token && getToken() === token) clearToken();
    setSession(guest);
    closeAuth();
    clearNotice();
    if (token) void endSession(token).catch(() => {});
  }, [closeAuth, clearNotice]);

  const checkError = useCallback((error: unknown, token: string) => {
    // The confirmed merge may revoke the old Telegram login before its replacement arrives.
    if (!invalidSession(error) || activeToken.current !== token || accountChange.current?.token === token) return;
    activeToken.current = null;
    if (getToken() === token) clearToken();
    setSession(guest);
    setNotice('Сессия истекла. Войдите снова.');
    openAuth();
  }, [openAuth]);

  const uploadAvatar = useCallback(async (file: File) => {
    const token = session.token;
    if (!token) return;
    try {
      const profileImageUrl = await saveAvatar(token, file);
      setSession((old) => old.status === 'authenticated' && old.token === token
        ? { ...old, profile: { ...old.profile, profileImageUrl } }
        : old);
    } catch (error) {
      checkError(error, token);
      throw error;
    }
  }, [session.token, checkError]);

  const refreshProfile = useCallback(async (signal?: AbortSignal) => {
    const token = session.token;
    if (!token) return null;
    try {
      const profile = await getProfile(token, signal);
      if (signal?.aborted || activeToken.current !== token) return null;
      setSession((old) => old.status === 'authenticated' && old.token === token ? { ...old, profile } : old);
      return profile;
    } catch (error) {
      if (!signal?.aborted) checkError(error, token);
      throw error;
    }
  }, [session.token, checkError]);

  return {
    ...session, authOpen, notice, openAuth, closeAuth, clearNotice, retry, acceptCredentials,
    logout, checkError, uploadAvatar, refreshProfile, changeAccount,
  };
}

import type { PlayerStatus, Track } from '../../types/track';

interface AudioCallbacks {
  onStatus: (status: PlayerStatus) => void;
  onProgress: (progress: number) => void;
  onDuration: (duration: number) => void;
  onEnded: () => void;
  onNext: () => void;
  onPrevious: () => void;
  onPlaybackBlocked?: () => void;
  onMediaPlay?: () => void;
  onMediaPause?: () => void;
  onMediaSeek?: (position: number) => void;
}

const mediaActions: MediaSessionAction[] = [
  'play', 'pause', 'stop', 'previoustrack', 'nexttrack', 'seekto', 'seekforward', 'seekbackward',
];

export class AudioEngine {
  private readonly audio = new Audio();
  private readonly mediaSession = navigator.mediaSession;
  private readonly listeners: Partial<Record<keyof HTMLMediaElementEventMap, EventListener>>;
  private hasTrack = false;
  private wantsPlayback = false;
  private requestId = 0;
  private pendingPosition: number | null = null;

  constructor(private readonly callbacks: AudioCallbacks, volume: number) {
    this.audio.preload = 'metadata';
    this.audio.volume = volume;
    this.listeners = {
      playing: () => {
        if (this.wantsPlayback && !this.audio.paused && this.audio.readyState >= 3) {
          this.setStatus('playing');
        }
      },
      pause: () => {
        if (this.hasTrack && this.audio.paused && !this.audio.ended && this.wantsPlayback) {
          this.pause();
        }
      },
      waiting: () => {
        if (this.wantsPlayback && !this.audio.paused && this.audio.readyState < 3) {
          this.setStatus('loading');
        }
      },
      timeupdate: () => this.updateProgress(),
      durationchange: () => {
        this.callbacks.onDuration(this.duration);
        if (this.pendingPosition !== null && this.duration) this.seekTo(this.pendingPosition);
        this.updateProgress();
      },
      ended: () => {
        if (!this.wantsPlayback || !this.audio.ended) return;
        this.requestId++;
        this.wantsPlayback = false;
        this.setStatus('paused');
        this.callbacks.onEnded();
      },
      error: () => {
        if (!this.hasTrack || !this.audio.error) return;
        this.requestId++;
        this.wantsPlayback = false;
        this.setStatus('error');
      },
    };
    for (const [event, listener] of Object.entries(this.listeners)) {
      this.audio.addEventListener(event, listener);
    }
  }

  private get duration() {
    const { duration } = this.audio;
    return Number.isFinite(duration) && duration > 0 ? duration : 0;
  }

  get currentTime() {
    return this.audio.currentTime;
  }

  load(track: Track, { position = 0, playing = true } = {}) {
    this.requestId++;
    this.wantsPlayback = false;
    this.hasTrack = true;
    this.pendingPosition = position > 0 ? position : null;
    this.audio.src = track.streamUrl;
    this.audio.load();
    this.callbacks.onProgress(0);
    this.callbacks.onDuration(0);
    this.setMediaTrack(track);
    if (playing) this.resume();
    else this.setStatus('paused');
  }

  togglePlayback() {
    if (this.wantsPlayback) this.pause();
    else this.resume();
  }

  restart() {
    this.seek(0);
    this.resume();
  }

  seek(ratio: number) {
    if (Number.isFinite(ratio)) this.seekTo(ratio * this.duration);
  }

  setVolume(volume: number) {
    this.audio.volume = volume;
  }

  dispose() {
    this.hasTrack = false;
    this.wantsPlayback = false;
    this.requestId++;
    for (const [event, listener] of Object.entries(this.listeners)) {
      this.audio.removeEventListener(event, listener);
    }
    this.audio.pause();
    this.audio.removeAttribute('src');
    this.audio.load();
    if (this.mediaSession) {
      this.mediaSession.metadata = null;
      this.mediaSession.playbackState = 'none';
      this.setMediaPosition();
      for (const action of mediaActions) this.setMediaAction(action, null);
    }
  }

  resume() {
    if (!this.hasTrack || this.wantsPlayback) return;
    if (this.audio.error) this.audio.load();
    this.wantsPlayback = true;
    const requestId = ++this.requestId;
    this.setStatus('loading');
    void this.audio.play().catch((error: unknown) => {
      if (requestId !== this.requestId) return;
      this.wantsPlayback = false;
      this.setStatus('error');
      if (error instanceof DOMException && error.name === 'NotAllowedError') this.callbacks.onPlaybackBlocked?.();
    });
  }

  pause() {
    if (!this.hasTrack) return;
    this.requestId++;
    this.wantsPlayback = false;
    this.audio.pause();
    this.setStatus('paused');
  }

  private setStatus(status: PlayerStatus) {
    this.callbacks.onStatus(status);
    if (this.mediaSession && status !== 'loading') {
      this.mediaSession.playbackState = status === 'playing' ? 'playing' : 'paused';
    }
  }

  seekTo(seconds: number) {
    if (!this.hasTrack || !Number.isFinite(seconds)) return;
    if (!this.duration) {
      this.pendingPosition = Math.max(0, seconds);
      return;
    }
    this.pendingPosition = null;
    this.audio.currentTime = Math.min(this.duration, Math.max(0, seconds));
    this.updateProgress();
  }

  private updateProgress() {
    const duration = this.duration;
    if (!this.hasTrack || !duration) return;
    const position = Math.min(duration, Math.max(0, this.currentTime));
    this.callbacks.onProgress(position / duration);
    this.setMediaPosition({ duration, position, playbackRate: this.audio.playbackRate });
  }

  private setMediaTrack(track: Track) {
    const session = this.mediaSession;
    if (!session) return;
    this.setMediaPosition();
    if (typeof MediaMetadata !== 'undefined') {
      try {
        session.metadata = new MediaMetadata({
          title: track.title,
          artist: track.artist,
          album: track.album,
          artwork: track.coverUrl ? [{ src: track.coverUrl }] : [],
        });
      } catch {
        session.metadata = null;
      }
    }
    const play = this.callbacks.onMediaPlay ?? (() => this.resume());
    const pause = this.callbacks.onMediaPause ?? (() => this.pause());
    const seek = this.callbacks.onMediaSeek ?? ((position: number) => this.seekTo(position));
    this.setMediaAction('play', play);
    this.setMediaAction('pause', pause);
    this.setMediaAction('stop', () => {
      pause();
      seek(0);
    });
    this.setMediaAction('previoustrack', () => this.callbacks.onPrevious());
    this.setMediaAction('nexttrack', () => this.callbacks.onNext());
    this.setMediaAction('seekto', ({ seekTime }) => {
      if (seekTime !== undefined) seek(seekTime);
    });
    this.setMediaAction('seekforward', ({ seekOffset }) => seek(this.currentTime + (seekOffset ?? 10)));
    this.setMediaAction('seekbackward', ({ seekOffset }) => seek(this.currentTime - (seekOffset ?? 10)));
  }

  private setMediaAction(action: MediaSessionAction, handler: MediaSessionActionHandler | null) {
    try {
      this.mediaSession?.setActionHandler(action, handler);
    } catch {
      // Action support varies by browser and operating system.
    }
  }

  private setMediaPosition(position?: MediaPositionState) {
    try {
      this.mediaSession?.setPositionState?.(position);
    } catch {
      // Some media sessions reject position updates while switching tracks.
    }
  }
}

function hex(bytes: Uint8Array) {
  return [...bytes].map((byte) => byte.toString(16).padStart(2, '0')).join('');
}

export function createPlaybackDevice(): DeviceCredentials {
  const uuid = crypto.getRandomValues(new Uint8Array(16));
  uuid[6] = (uuid[6] & 0x0f) | 0x40;
  uuid[8] = (uuid[8] & 0x3f) | 0x80;
  const text = hex(uuid);
  const deviceId = [text.slice(0, 8), text.slice(8, 12), text.slice(12, 16), text.slice(16, 20), text.slice(20)].join('-');
  return { deviceId, deviceKey: hex(crypto.getRandomValues(new Uint8Array(32))) };
}

function isNewer(snapshot: PlaybackSnapshot, previous: PlaybackSnapshot | null) {
  if (!previous) return true;
  if (snapshot.session.version !== previous.session.version) return snapshot.session.version > previous.session.version;
  const lease = snapshot.session.leaseUntil ?? 0;
  const previousLease = previous.session.leaseUntil ?? 0;
  if (lease !== previousLease) return lease > previousLease;
  const anchor = snapshot.session.positionAt ?? 0;
  const previousAnchor = previous.session.positionAt ?? 0;
  if (anchor !== previousAnchor) return anchor > previousAnchor;
  return snapshot.serverTime >= previous.serverTime;
}

function leaseRemaining(snapshot: PlaybackSnapshot) {
  return (snapshot.session.leaseUntil ?? 0) - snapshot.serverTime;
}

export class PlaybackSession {

  private readonly listeners = new Set<() => void>();

  private leaseTimer = 0;
  private claimTimer = 0;
  private heartbeatTimer = 0;
  private progressTimer = 0;
  private reconnectTimer = 0;
  private seekTimer = 0;
  private engine: AudioEngine | null = null;
  private engineTrackId: number | null = null;
  private engineStatus: PlayerStatus = 'idle';
  private engineDuration = 0;
  private needsGesture = false;
  private registered = false;
  private refreshInFlight: Promise<void> | null = null;

  private get deviceId() { return this.credentials.deviceId; }

  subscribe = (listener: () => void) => {

  getView = () => this.view;

  start() {

    on('pagehide', this.suspend);

  retryPlayback = () => {

  private queueSeek(position: number) {

    if (trackId == null || !this.view.duration || !Number.isFinite(position)) return;
    this.deferredSeek = { trackId, position: Math.max(0, Math.min(this.view.duration, position)) };
    if (this.hasLease()) this.engine?.seekTo(this.deferredSeek.position);
    this.publish();
    clearTimeout(this.seekTimer);
    this.seekTimer = window.setTimeout(() => {
      const seek = this.deferredSeek;
      this.deferredSeek = null;
      if (seek) this.send(({ session }) => session.trackId === seek.trackId

    }, 120);
  }

  private pauseCommand(): PlaybackCommand {
    return { type: 'pause', position: this.hasLease() ? this.engine?.currentTime : undefined };
  }

  private get serverNow() {
    return (this.snapshot?.serverTime ?? 0) + (performance.now() - this.receivedAt);
  }

  private isPlaying({ session }: PlaybackSnapshot) {
    return session.playing && !session.pendingDeviceId && (session.leaseUntil ?? 0) > this.serverNow;
  }

  private hasLease() {
    const session = this.snapshot?.session;
    return this.running && session?.activeDeviceId === this.deviceId && !session.pendingDeviceId &&
      performance.now() < this.leaseDeadline;
  }

  private position() {
    const snapshot = this.snapshot;
    if (!snapshot) return 0;
    const { session } = snapshot;
    const limit = Math.min(this.serverNow, session.leaseUntil ?? snapshot.serverTime);
    const elapsed = session.playing && !session.pendingDeviceId ? Math.max(0, limit - snapshot.serverTime) / 1_000 : 0;
    return Math.max(0, session.position + elapsed);
  }

  private checkLease = () => {
    if (this.engine && !this.hasLease()) this.stopAudio();
    if (!document.hidden && this.running) void this.refresh();
    this.publish();
  };

  private tick = () => {
    if (!this.running) return;
    if (this.engine && !this.hasLease()) this.stopAudio();
    if (this.snapshot && !this.engine && this.snapshot.session.playing) this.publish();
  };

  private suspend = () => {
    if (this.suspended) return;
    const hasLease = this.hasLease();
    const session = this.snapshot?.session;
    const position = this.stopAudio(session?.trackId);
    this.suspended = true;
    this.generation++;
    this.leaseDeadline = 0;
    this.deferredSeek = null;
    clearTimeout(this.leaseTimer);
    clearTimeout(this.claimTimer);
    clearTimeout(this.seekTimer);
    clearTimeout(this.reconnectTimer);
    this.eventsAbort?.abort();
    if (hasLease && session) {
      void this.api.command(session.version, { type: 'release', position }, undefined, true).catch(() => {});
    }
  };

  private wake = () => {
    if (!this.suspended || this.abort.signal.aborted || this.connection === 'unsupported') return;
    this.suspended = false;
    this.refreshSoon();
    void this.connectEvents();
  };

  private refresh(): Promise<void> {
    if (!this.running) return Promise.resolve();
    if (this.refreshInFlight) return this.refreshInFlight;
    this.refreshInFlight = (async () => {
      let registrationError: unknown;
      if (!this.registered) {
        registrationError = await this.request((signal) => this.api.register(this.deviceLabel, this.platform, signal), false, () => {
          this.registered = true;
        });
      }
      if (this.registered) {
        const report = this.getReport();
        const error = await this.request((signal) => this.api.heartbeat(signal, report), true, () => {
          if (report) this.lastReportAt = performance.now();
        });
        if (error instanceof PlaybackApiError && (error.status === 403 || error.status === 404)) {
          this.registered = false;
        }
      } else if (!this.snapshot) {
        const sessionError = await this.request((signal) => this.api.session(signal), false);
        // A legacy server has neither route. A live shared session or a transient error
        // must still use the lease rather than starting a second local audio engine.
        if (this.running && !this.snapshot &&
            registrationError instanceof PlaybackApiError && registrationError.status === 404 &&
            sessionError instanceof PlaybackApiError && sessionError.status === 404) {
          this.suspend();
          clearInterval(this.heartbeatTimer);
          clearInterval(this.progressTimer);
          this.connection = 'unsupported';
          this.notice = null;
          this.publish();
        }
      }
    })().finally(() => {
      this.refreshInFlight = null;
      if (this.refreshRequested) {
        this.refreshRequested = false;
        void this.refresh();
      }
    });
    return this.refreshInFlight;
  }

  private getReport() {
    const session = this.snapshot?.session;
    if (!session || session.trackId === null || !this.hasLease() || !this.engine ||
      this.engineTrackId !== session.trackId || this.engineStatus !== 'playing' ||
      performance.now() - this.lastReportAt < 12_000) return undefined;
    return { trackId: session.trackId, version: session.version, position: this.engine.currentTime };
  }

  private refreshSoon() {
    if (this.refreshInFlight) this.refreshRequested = true;
    else void this.refresh();
  }

  private async request(operation: Operation, confirmsLease: boolean, onSuccess?: () => void) {
    if (!this.running) return;
    const controller = new AbortController();
    const cancel = () => controller.abort();
    const timeout = window.setTimeout(cancel, 6_000);
    this.abort.signal.addEventListener('abort', cancel, { once: true });
    const startedAt = performance.now();
    const generation = this.generation;
    try {
      const snapshot = await operation(controller.signal);
      if (!this.running || generation !== this.generation) return;
      onSuccess?.();
      this.acceptSnapshot(snapshot, confirmsLease ? startedAt : undefined);
    } catch (error) {
      if (!this.running || generation !== this.generation) return;
      this.handleError(error);
      return error;
    } finally {
      clearTimeout(timeout);
      this.abort.signal.removeEventListener('abort', cancel);
    }
  }

  private handleError(error: unknown) {
    if (error instanceof PlaybackApiError && error.status === 401) {
      this.suspend();
      this.onSessionExpired(error, this.token);
      return;
    }
    if (error instanceof PlaybackApiError && error.snapshot) {
      this.acceptSnapshot(error.snapshot);
      return;
    }
    this.connection = this.snapshot ? 'reconnecting' : 'offline';
    this.publish();
  }

  private async connectEvents() {
    if (!this.running || this.eventsAbort) return;
    const controller = new AbortController();
    this.eventsAbort = controller;
    let idleTimer = 0;
    const activity = () => {
      clearTimeout(idleTimer);
      idleTimer = window.setTimeout(() => controller.abort(), 35_000);
    };
    activity();
    try {
      await this.api.events(controller.signal, (snapshot) => {
        if (!this.running || controller.signal.aborted) return;
        this.reconnectDelay = 1_000;
        this.acceptSnapshot(snapshot);
      }, activity);
    } catch (error) {
      // Heartbeats deliver snapshots while the event stream is unavailable.
      if (this.running && error instanceof PlaybackApiError && error.status === 401) this.handleError(error);
    } finally {
      clearTimeout(idleTimer);
      this.eventsAbort = null;
      if (this.running) {
        this.reconnectTimer = window.setTimeout(() => void this.connectEvents(), this.reconnectDelay);
        this.reconnectDelay = Math.min(15_000, this.reconnectDelay * 2);
      }
    }
  }

  private send(createCommand: (snapshot: PlaybackSnapshot) => PlaybackCommand | null) {
    const generation = this.generation;
    this.commands = this.commands.catch(() => {}).then(async () => {
      if (!this.running || generation !== this.generation) return;
      if (!this.registered || !this.snapshot) await this.refresh();
      if (!this.running || generation !== this.generation) return;
      if (!this.registered || !this.snapshot) {
        this.notice = 'Не удалось связаться с плеером. Попробуйте ещё раз.';
        this.publish();
        return;
      }
      const command = createCommand(this.snapshot);
      if (!command) return;
      const version = this.snapshot.session.version;
      const error = await this.request((signal) => this.api.command(version, command, signal), true);
      if (!this.running || generation !== this.generation) return;
      this.notice = error ? 'Не удалось применить действие. Попробуйте ещё раз.' : null;
      this.publish();
    });
  }

  private acceptSnapshot(snapshot: PlaybackSnapshot, startedAt?: number) {
    if (!Number.isSafeInteger(snapshot?.session?.version) || !Number.isFinite(snapshot?.serverTime) ||
      !Array.isArray(snapshot.session.queue) || !Array.isArray(snapshot.devices)) return;
    const previous = this.snapshot;
    if (isNewer(snapshot, previous)) {
      this.snapshot = snapshot;
      this.receivedAt = performance.now();
    }
    const current = this.snapshot!;
    const { session } = current;
    const own = session.activeDeviceId === this.deviceId;
    const handoff = own && session.pendingDeviceId && (session.leaseUntil ?? 0) > this.serverNow;
    const stillActive = own && !session.pendingDeviceId && (session.leaseUntil ?? 0) > this.serverNow;
    this.connection = 'online';

    if (handoff) {
      this.ackPosition = this.stopAudio(session.trackId) ?? this.ackPosition;
      this.leaseDeadline = 0;
      this.acknowledge(current);
    } else if (!stillActive) {
      this.stopAudio();
      this.leaseDeadline = 0;
      this.ackPosition = undefined;
    } else {
      const confirmed = startedAt !== undefined && snapshot.session.version === session.version &&
        snapshot.session.activeDeviceId === this.deviceId && !snapshot.session.pendingDeviceId;
      if (confirmed) {
        const deadline = startedAt + leaseRemaining(snapshot) - deadlineMargin;
        this.leaseDeadline = Math.max(this.leaseDeadline, deadline);
        clearTimeout(this.leaseTimer);
        this.leaseTimer = window.setTimeout(this.checkLease, Math.max(0, this.leaseDeadline - performance.now()));
      } else if (previous?.session.activeDeviceId !== this.deviceId || previous.session.pendingDeviceId) {
        this.refreshSoon();
      }
      this.ackPosition = undefined;
    }

    clearTimeout(this.claimTimer);
    if (session.pendingDeviceId === this.deviceId) {
      const wait = Math.max(0, (session.leaseUntil ?? 0) - this.serverNow);
      // Retry promptly after an unresponsive previous device's lease expires.
      this.claimTimer = window.setTimeout(() => this.refreshSoon(), Math.max(250, wait + 100));
    }

    this.resolveTracks();
    this.applyAudio(previous?.session.version !== session.version);
    this.publish();
  }

  private acknowledge({ session }: PlaybackSnapshot) {
    if (this.ackInFlight) return;
    this.ackInFlight = true;
    void this.request((signal) => this.api.acknowledge(session.version, this.ackPosition, signal), true)
      .finally(() => { this.ackInFlight = false; });
  }

  private resolveTracks() {
    const session = this.snapshot?.session;
    if (!session) return;
    const missing = (session.trackId !== null && !this.tracks.has(session.trackId)) ||
      session.queue.some((id) => !this.tracks.has(id));
    if (!missing || performance.now() < this.nextCatalogAttempt) return;
    this.nextCatalogAttempt = performance.now() + 8_000;
    void getTracks(this.abort.signal).then((tracks) => {
      if (!this.running) return;
      for (const track of tracks) this.tracks.set(track.id, track);
      this.applyAudio();
      this.publish();
    }).catch(() => {});
  }

  private applyAudio(timelineChanged = false) {
    const snapshot = this.snapshot;
    if (!snapshot || !this.hasLease()) {
      this.stopAudio();
      return;
    }
    const { trackId, version } = snapshot.session;
    const track = trackId === null ? undefined : this.tracks.get(trackId);
    if (!track) {
      this.stopAudio();
      return;
    }
    const playing = this.isPlaying(snapshot) && this.endedVersion !== version;
    const position = this.deferredSeek?.trackId === track.id ? this.deferredSeek.position : this.position();
    const engine = this.engine ?? this.createEngine();
    if (this.engineTrackId !== track.id) {
      this.engineTrackId = track.id;
      this.needsGesture = false;
      engine.load(track, { position, playing });
      return;
    }
    const acceptableDrift = timelineChanged ? 0.1 : playing ? 1.5 : 0.25;
    if (Math.abs(engine.currentTime - position) > acceptableDrift) engine.seekTo(position);
    if (!playing && this.engineStatus !== 'paused') {
      engine.pause();
    } else if (playing && !this.needsGesture && this.engineStatus !== 'error') engine.resume();
  }

  private createEngine() {
    const engine = new AudioEngine({
      onStatus: (status) => {
        if (this.engine !== engine) return;
        this.engineStatus = status;
        this.publish();
      },
      onDuration: (duration) => {
        if (this.engine !== engine) return;
        this.engineDuration = duration;
        this.publish();
      },
      onProgress: () => { if (this.engine === engine) this.publish(); },
      onEnded: () => {
        if (this.engine !== engine || !this.hasLease() || !this.snapshot || this.engineTrackId === null) return;
        const trackId = this.engineTrackId;
        const version = this.snapshot.session.version;
        this.endedVersion = version;
        this.send(({ session }) => this.hasLease() && session.version === version ? { type: 'ended', trackId } : null);
      },
      onPlaybackBlocked: () => {
        if (this.engine !== engine) return;
        this.needsGesture = true;
        const trackId = this.engineTrackId;
        const position = this.engineDuration ? engine.currentTime : this.position();
        this.send((snapshot) => this.hasLease() && snapshot.session.trackId === trackId && this.isPlaying(snapshot)
          ? { type: 'pause', position } : null);
        this.publish();
      },
      onMediaPlay: () => {
        if (this.needsGesture) this.retryPlayback();
        else this.send(() => ({ type: 'play' }));
      },
      onMediaPause: () => this.send(() => this.pauseCommand()),
      onMediaSeek: (position) => this.queueSeek(position),
      onNext: this.playNext,
      onPrevious: this.playPrev,
    }, this.volume);
    this.engine = engine;
    return engine;
  }

  private stopAudio(trackId?: number | null) {
    if (!this.engine) return undefined;
    const engine = this.engine;
    const position = trackId === undefined || this.engineTrackId === trackId ? engine.currentTime : undefined;
    this.engine = null;
    this.engineTrackId = null;
    this.engineStatus = 'idle';
    this.engineDuration = 0;
    this.needsGesture = false;
    this.endedVersion = null;
    engine.dispose();
    return position;
  }

  private publish() {
    const snapshot = this.snapshot;
    const session = snapshot?.session;
    const trackId = session?.trackId;
    const currentTrack = trackId == null ? null : this.tracks.get(trackId) ?? null;
    const resolvedQueue = (session?.queue ?? []).flatMap((id) => {
      const track = this.tracks.get(id);
      return track ? [track] : [];
    });
    const previousQueue = this.view.queue;
    const queue = previousQueue.length === resolvedQueue.length && previousQueue.every((track, index) => track === resolvedQueue[index])
      ? previousQueue : resolvedQueue;
    const local = this.hasLease();
    const duration = (local ? this.engineDuration : 0) || session?.duration || currentTrack?.duration || 0;
    const position = this.deferredSeek && this.deferredSeek.trackId === trackId
      ? this.deferredSeek.position : local && this.engine ? this.engine.currentTime : this.position();
    const remotePlaying = snapshot && this.isPlaying(snapshot);

    this.view = {
      currentTrack, queue, queueIndex: queue.findIndex((track) => track.id === trackId),
      status: currentTrack ? local && this.engine ? this.engineStatus : remotePlaying ? 'playing' : 'paused' : 'idle',
      progress: duration ? Math.min(1, Math.max(0, position / duration)) : 0,
      duration, shuffle: session?.shuffle ?? false, repeat: session?.repeat ?? 'none',
      mode: local ? 'active' : 'remote', connection: this.connection, devices: snapshot?.devices ?? [],
      activeDeviceId: (session?.leaseUntil ?? 0) > this.serverNow ? session?.activeDeviceId ?? null : null,
      pendingDeviceId: session?.pendingDeviceId ?? null, needsGesture: this.needsGesture, notice: this.notice,
    };
    for (const listener of this.listeners) listener();
  }
}

          {(auth.notice || favorites.notice) && (

              auth.clearNotice();

          )}

      )}

        className="flex-1 overflow-y-auto"

          </div>
        ) : rows.length === 0 ? (
          <div className="flex flex-col items-center gap-2 py-20 text-muted">
            <span className="text-[17px] font-semibold text-text">
              {queueTab ? 'Queue is empty' : favoriteTab ? 'No favorites yet' : query ? 'No results' : 'No tracks'}
            </span>
            {favoriteTab && <p className="px-8 text-center text-sm">Tap ♡ on any track to save it here</p>}
          </div>
        ) : (

          onChangeVolume={player.changeVolume}
          onToggleShuffle={player.toggleShuffle}
          onCycleRepeat={player.cycleRepeat}
        />
      )}

      <nav
        aria-label="Music navigation"
        className="fixed bottom-0 left-0 right-0 z-[101] flex border-t border-[var(--border)] bg-surface/95 backdrop-blur-xl"
        style={{ paddingBottom: 'env(safe-area-inset-bottom)', height: 'calc(56px + env(safe-area-inset-bottom))' }}
      >
        <TabBtn icon={<HomeIcon active={tab === 'all'} />} label="Music" active={tab === 'all'}
          onClick={() => { hapticTap(); setTab('all'); }} />
        <TabBtn icon={<HeartTabIcon active={favoriteTab} count={favorites.favoriteIds.size} />}
          label="Favorites" active={favoriteTab} onClick={() => { hapticTap(); setTab('favorites'); }} />
        <TabBtn icon={<Icon name="queue" size={22} />} label="Queue" active={queueTab}
          onClick={() => { hapticTap(); setTab('queue'); }} />
      </nav>

      {profile ? (
        <ProfilePanel
          open={auth.authOpen}
          profile={profile}
          onClose={auth.closeAuth}
          onLogout={auth.logout}
          onUploadAvatar={auth.uploadAvatar}

          open={auth.authOpen}

          onLogout={auth.logout}
          onUploadAvatar={auth.uploadAvatar}
          onCredentials={auth.acceptCredentials}
        />
      )}
    </div>
  );
}

export const BASE_URL = (
  import.meta.env.VITE_API_BASE_URL || 'https://music-streaming-server-lfon.onrender.com'
).replace(/\/$/, '');
const TOKEN_KEY = 'tg_music_token';

export const getToken = () => localStorage.getItem(TOKEN_KEY);
export const setToken = (token: string) => localStorage.setItem(TOKEN_KEY, token);
export const clearToken = () => localStorage.removeItem(TOKEN_KEY);

export class ApiError extends Error {
  constructor(readonly status: number, message: string) {
    super(message);
  }
}

export interface Profile {
  id: number;
  username: string | null;
  displayName?: string | null;
  telegramName?: string | null;
  telegramId?: string | null;
  profileImageUrl?: string | null;
  bio?: string | null;
  email?: string | null;
  createdAt?: string;
  isAdmin?: boolean;
  hasPassword?: boolean;
}

function mediaUrl(value?: string | null) {
  if (!value) return undefined;
  try {
    const base = new URL(`${BASE_URL}/`, window.location.href);
    const url = new URL(value.replace(/^\/+/, ''), base);
    if (url.protocol !== 'http:' && url.protocol !== 'https:') return undefined;
    if (url.protocol === 'http:' && window.location.protocol === 'https:') url.protocol = 'https:';
    return url.href;
  } catch {
    return undefined;
  }
}

async function request<T>(path: string, options?: RequestInit, token?: string): Promise<T> {
  const headers = new Headers(options?.headers);
  if (token) headers.set('Authorization', `Bearer ${token}`);

  const response = await fetch(`${BASE_URL}${path}`, { ...options, headers });
  if (!response.ok) {
    const body = await response.json().catch(() => null);
    throw new ApiError(response.status, body?.error ?? `API error ${response.status}`);
  }
  return response.json();
}

function json(method: string, body: object, signal?: AbortSignal): RequestInit {
  return {
    method,
    signal,
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  };
}

async function requestToken(path: string, body: object, signal?: AbortSignal) {
  const result = await request<{ token?: string }>(path, json('POST', body, signal));
  if (!result.token) throw new Error('Сервер не вернул данные для входа.');
  return result.token;
}

export async function getProfile(token: string, signal?: AbortSignal): Promise<Profile> {
  const profile = await request<Profile>('/users/profile', { signal }, token);
  return { ...profile, profileImageUrl: mediaUrl(profile.profileImageUrl) };
}

  return request('/users/logout', { method: 'POST' }, token);
}

export async function uploadAvatar(token: string, file: File) {
  const body = new FormData();
  body.append('profileImage', file);
  const result = await request<{ profileImageUrl: string }>(
    '/users/uploadProfileImage',
    { method: 'POST', body },
    token,
  );
  return mediaUrl(result.profileImageUrl);
}

interface RawTrack {
  id: number;
  title?: string | null;
  artist?: string | null;
  imageUrl?: string | null;
  album?: string | null;
  genre?: string | null;
  duration?: number | null;
}

function toTrack(raw: RawTrack): Track {
  return {
    id: raw.id,
    title: raw.title || 'Unknown',
    artist: raw.artist || 'Unknown',
    album: raw.album ?? undefined,
    genre: raw.genre ?? undefined,
    duration: raw.duration ?? undefined,
    coverUrl: mediaUrl(raw.imageUrl),
    streamUrl: `${BASE_URL}/tracks/${raw.id}/stream`,
  };
}

export async function getTracks(signal?: AbortSignal) {
  return (await request<RawTrack[]>('/tracks', { signal })).map(toTrack);
}

export async function searchTracks(query: string, signal: AbortSignal) {
  const path = `/tracks/search?query=${encodeURIComponent(query)}`;
  return (await request<RawTrack[]>(path, { signal })).map(toTrack);
}

export async function getFavorites(token: string, signal: AbortSignal) {
  type Favorite = Omit<RawTrack, 'id'> & { trackId: number };
  const { favorites } = await request<{ favorites: Favorite[] }>('/favorites/get', { signal }, token);
  return favorites.map((favorite) => toTrack({ ...favorite, id: Number(favorite.trackId) }));
}

export function toggleFavorite(token: string, trackId: number, isFavorite: boolean) {
  const path = isFavorite ? '/favorites/remove' : '/favorites/add';
  return request(path, json('POST', { trackId }), token);
}

import { useCallback, useLayoutEffect, useMemo, useRef, useState, useSyncExternalStore } from 'react';
import type { DevicePlatform } from '../../services/playback';
import type { PlayerStatus, Track } from '../../types/track';
import { AudioEngine } from './AudioEngine';
import { createPlaybackDevice, initialPlaybackView, PlaybackSession } from './PlaybackSession';
import {
  cycleRepeat, emptyQueue, nextTrack, previousTrack, selectTrack, toggleShuffle,
  type PlaybackQueue,
} from './PlayerStore';

const volumeKey = 'tg_music_volume';

function savedVolume() {
  const stored = localStorage.getItem(volumeKey);
  const value = stored === null ? 1 : Number(stored);
  return Number.isFinite(value) ? Math.min(1, Math.max(0, value)) : 1;
}

function useLocalPlayer(enabled: boolean) {
  const [queue, setQueue] = useState(emptyQueue);
  const [status, setStatus] = useState<PlayerStatus>('idle');
  const [progress, setProgress] = useState(0);
  const [duration, setDuration] = useState(0);
  const [volume, setVolume] = useState(savedVolume);
  const queueRef = useRef(queue);
  const volumeRef = useRef(volume);
  const engineRef = useRef<AudioEngine | null>(null);

  const updateQueue = useCallback((next: PlaybackQueue) => {
    queueRef.current = next;
    setQueue(next);
  }, []);

  const load = useCallback((next: PlaybackQueue | null) => {
    const engine = engineRef.current;
    if (!engine || !next) return;
    updateQueue(next);
    engine.load(next.tracks[next.index]);
  }, [updateQueue]);

  const playNext = useCallback(() => {
    load(nextTrack(queueRef.current, false));
  }, [load]);

  const playPrev = useCallback(() => {
    const engine = engineRef.current;
    if (!engine) return;
    if (engine.currentTime > 3) {
      engine.seek(0);
      return;
    }
    load(previousTrack(queueRef.current));
  }, [load]);

  useLayoutEffect(() => {
    if (!enabled) {
      updateQueue(emptyQueue);
      setStatus('idle');
      setProgress(0);
      setDuration(0);
      return;
    }
    const engine = new AudioEngine({
      onStatus: setStatus,
      onProgress: setProgress,
      onDuration: setDuration,
      onEnded: () => {
        if (queueRef.current.repeat === 'one') engine.restart();
        else load(nextTrack(queueRef.current, true));
      },
      onPrevious: playPrev,
      onNext: playNext,
    }, volumeRef.current);
    engineRef.current = engine;
    return () => {
      engineRef.current = null;
      engine.dispose();
    };
  }, [enabled, load, playNext, playPrev, updateQueue]);

  const play = useCallback((track: Track, trackList?: Track[]) => {
    const currentQueue = queueRef.current;
    if (currentQueue.tracks[currentQueue.index]?.id === track.id) {
      engineRef.current?.togglePlayback();
      return;
    }
    load(selectTrack(currentQueue, track, trackList));
  }, [load]);

  const seek = useCallback((ratio: number) => engineRef.current?.seek(ratio), []);

  const changeVolume = useCallback((value: number) => {
    if (!Number.isFinite(value)) return;
    const next = Math.min(1, Math.max(0, value));
    engineRef.current?.setVolume(next);
    volumeRef.current = next;
    setVolume(next);
    localStorage.setItem(volumeKey, String(next));
  }, []);

  const handleShuffle = useCallback(() => updateQueue(toggleShuffle(queueRef.current)), [updateQueue]);
  const handleRepeat = useCallback(() => updateQueue(cycleRepeat(queueRef.current)), [updateQueue]);

  return {
    currentTrack: queue.tracks[queue.index] ?? null,
    queue: queue.tracks,
    queueIndex: queue.index,
    status,
    progress,
    duration,
    volume,
    shuffle: queue.shuffle,
    repeat: queue.repeat,
    play,
    playNext,
    playPrev,
    seek,
    changeVolume,
    toggleShuffle: handleShuffle,
    cycleRepeat: handleRepeat,
  };
}

const noop = () => {};
const emptyCatalog: readonly Track[] = [];
const initialView = () => initialPlaybackView;
const subscribeIdle = () => noop;

export interface PlaybackOptions {
  token?: string | null;
  userId?: number | null;
  deviceLabel?: string;
  platform?: DevicePlatform;
  catalog?: readonly Track[];
  onSessionExpired?: (error: unknown, token: string) => void;
}

export function usePlayerController({
  token = null, userId = null, deviceLabel = 'Web', platform = 'browser',
  catalog = emptyCatalog, onSessionExpired,
}: PlaybackOptions) {
  const authenticated = token !== null && userId !== null;
  const credentials = useMemo(createPlaybackDevice, [userId]);
  const [connection, setConnection] = useState<{
    credentials: typeof credentials;
    session: PlaybackSession;
  } | null>(null);
  const synced = connection?.credentials === credentials && connection.session.token === token ? connection.session : null;
  const view = useSyncExternalStore(synced?.subscribe ?? subscribeIdle, synced?.getView ?? initialView);
  const legacyApi = authenticated && view.connection === 'unsupported';
  const local = useLocalPlayer(!authenticated || legacyApi);
  const latest = useRef({ catalog, volume: local.volume, onSessionExpired });
  useLayoutEffect(() => {
    latest.current = { catalog, volume: local.volume, onSessionExpired };
  }, [catalog, local.volume, onSessionExpired]);

  useLayoutEffect(() => {
    if (token === null || userId === null) {
      setConnection(null);
      return;
    }
    const session = new PlaybackSession(token, credentials, deviceLabel, platform, latest.current.volume,
      (error, expiredToken) => latest.current.onSessionExpired?.(error, expiredToken));
    session.rememberCatalog(latest.current.catalog);
    session.start();
    setConnection({ credentials, session });
    return () => session.dispose();
  }, [credentials, deviceLabel, platform, token, userId]);

  useLayoutEffect(() => { synced?.rememberCatalog(catalog); }, [catalog, synced]);
  useLayoutEffect(() => { synced?.setVolume(local.volume); }, [local.volume, synced]);

  if (!authenticated || legacyApi) return {
    ...initialPlaybackView,
    ...local,
    mode: 'local' as const,
    connection: legacyApi ? view.connection : 'idle' as const,
    deviceId: credentials.deviceId,
    takeOver: noop,
    clearNotice: noop,
    retryPlayback: noop,
  };

  return {
    ...view,
    deviceId: credentials.deviceId,
    volume: local.volume,
    changeVolume: local.changeVolume,
    play: synced?.play ?? noop,
    playNext: synced?.playNext ?? noop,
    playPrev: synced?.playPrev ?? noop,
    toggleShuffle: synced?.toggleShuffle ?? noop,
    cycleRepeat: synced?.cycleRepeat ?? noop,
    seek: synced?.seek ?? noop,
    takeOver: synced?.takeOver ?? noop,
    clearNotice: synced?.clearNotice ?? noop,
    retryPlayback: synced?.retryPlayback ?? noop,
  };
}

# tg-music-app

React, TypeScript and Vite power the desktop site, mobile site and Telegram Mini App. The layout changes at 651 px. Authentication, catalog, search, favorites and the active audio player are shared across layouts.

## Local development

```powershell
npm ci
Copy-Item .env.example .env.local
npm run dev
```

Start the Express backend on port 4000 or set `VITE_API_BASE_URL` to your deployed API. `npm run dev:desktop` opens the same responsive app at `127.0.0.1:5180` and proxies `/backend` through Vite.

```powershell
npm run build
```

Deploy `dist/` to your static host. Set `VITE_API_BASE_URL` to the API's HTTPS URL and add the frontend origin to the backend's `CORS_ORIGINS` if it differs from the default. To open the Mini App from Telegram, set the same HTTPS URL in the bot's menu button in [BotFather](https://t.me/BotFather).

Set `VITE_SUPABASE_URL` and `VITE_SUPABASE_ANON_KEY` to enable Realtime updates of the public catalog. The browser uses a public key for this subscription. The backend uses a separate service key for Storage.

## Listen on multiple devices

In the website profile, generate a Telegram link code, then enter and confirm it in the Mini App profile. Both sign-ins then use the same account and queue.

Start a track on the website, then open the Mini App on the linked account to control it. Open the device picker from the desktop player or the mobile header to move audio to another connected device. Tap **Continue playback here** if the receiving browser or Telegram requires interaction to start audio. Guests can play music locally.

## Project structure

- `src/app` starts the app and owns the shared player provider.
- `src/layouts` contains the responsive shell and desktop and mobile layouts.
- `src/features` contains auth, profile, player, library, search and favorites logic.
- `src/components` contains shared UI, including track rows, artwork and search.
- `src/platform/telegram` initializes the Telegram SDK and handles theme and haptics.
- `src/services` contains the backend API client and Realtime subscription.

          }
        : undefined,
    },
    build: { outDir: 'dist', sourcemap: false },
